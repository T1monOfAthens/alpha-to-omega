P3373 【模板】线段树 2

首先，我们考虑第一件事，把两个标记分开存。

我们开 add[p],mul[p] 分别存两个标记。

假设我们现在发生了两个操作：+3,×2+3,\times 2+3,×2。

(x+3)×2=2x+6(x+3) \times 2 = 2x + 6(x+3)×2=2x+6

而不是 2x+32x+32x+3。

注意到一个事情，就是乘法还会改变之前欠着的加法。

我们直接考虑把两个标记写成一个式子：

x→x×mul+addx \to x \times mul + addx→x×mul+add

这里我们要注意一个事情：x→x×1+0x \to x \times 1 + 0x→x×1+0，所以在什么都没有发生的时候，或者是我们清空标记的时候：

mul[p] = 1;
add[p] = 0;

加法​

x→x×m+ax→x×m+a+kx \to x \times m + a\\ x \to x \times m + a + kx→x×m+ax→x×m+a+k

这俩式子是不是就是把加法标记从 aaa 变成了 (a+k)(a+k)(a+k)。所以我们只需要修改加法标记就行了：

add[p] += k;

乘法​

在乘一个 kkk：

(x×m+a)×kx×(mk)+ak(x \times m + a) \times k \\ x\times(mk) + ak(x×m+a)×kx×(mk)+ak

所以我们发现，乘法标记被乘 kkk，加法标记也被乘 kkk。

所以：

tree[p] = tree[p] * k % MOD;
mul[p] = mul[p] * k % MOD;
add[p] = add[p] * k % MOD;

合并懒标记​

假设儿子有自己的标记：

x→x×mc+acx \to x \times m_c + a_cx→x×mc​+ac​

父亲还有标记：

x→x×mp+apx \to x \times m_p + a_px→x×mp​+ap​

如果这个时候，父亲要下放标记了，会发生什么。

注意操作顺序：儿子欠的操作先发生，父亲后来的操作再发生：

(xmc+ac)mp+ap(xm_c + a_c)m_p + a_p(xmc​+ac​)mp​+ap​

整理一下这个式子：

x(mcmp)+(acmp+ap)x(m_cm_p) + (a_cm_p + a_p)x(mc​mp​)+(ac​mp​+ap​)

所以合并之后：mc=mc×mp,ac=ac×mp+apm_c = m_c \times m_p,a_c = a_c \times m_p + a_pmc​=mc​×mp​,ac​=ac​×mp​+ap​。

注意​

线段树假设你要维护 nnn 个元素，数组长度开四倍。

#include<bits/stdc++.h>
using namespace std;

#define int long long

const int N=1e5+5;
int n,m,mod;
int a[N];
int tree[N*4];
int add[N*4];
int mul[N*4];
void pushUp(int p) {
    tree[p] = (tree[p * 2] + tree[p * 2 + 1]) % mod;
}
void build(int p,int l,int r) {
    mul[p] = 1;
    add[p] = 0;
    if (l == r) {
        tree[p] = a[l] % mod;
        return;
    }
    int mid = (l + r) / 2;
    build(p * 2, l, mid);
    build(p * 2 + 1, mid + 1, r);
    pushUp(p);
}
void pushDown(int p,int l,int r) {
    if (mul[p] == 1 && add[p] == 0) return;
    int mid = (l + r) / 2;
    int ls = p * 2;
    int rs = p * 2 + 1;
    int lenL = mid - l + 1;
    int lenR = r - mid;
    // 先修改两个儿子的区间和
    tree[ls] = (tree[ls] * mul[p] + add[p] * lenL) % mod;
    tree[rs] = (tree[rs] * mul[p] + add[p] * lenR) % mod;
    // 再合并两个儿子的 lazy 标记
    mul[ls] = mul[ls] * mul[p] % mod;
    mul[rs] = mul[rs] * mul[p] % mod;
    add[ls] = (add[ls] * mul[p] + add[p]) % mod;
    add[rs] = (add[rs] * mul[p] + add[p]) % mod;
    // 父亲标记清空
    mul[p] = 1;
    add[p] = 0;
}
// 区间乘 k
void updateMul(int p,int l,int r,int L,int R,int k) {
    if (L <= l && r <= R) {
        tree[p] = tree[p] * k % mod;
        mul[p] = mul[p] * k % mod;
        add[p] = add[p] * k % mod;
        return;
    }
    pushDown(p, l, r);
    int mid = (l + r) / 2;
    if (L <= mid) {
        updateMul(p * 2, l, mid, L, R, k);
    }
    if (R > mid) {
        updateMul(p * 2 + 1, mid + 1, r, L, R, k);
    }
    pushUp(p);
}
// 区间加 k
void updateAdd(int p,int l,int r,int L,int R,int k) {
    if (L <= l && r <= R) {
        tree[p] = (tree[p] + k * (r - l + 1)) % mod;
        add[p] = (add[p] + k) % mod;
        return;
    }
    pushDown(p, l, r);
    int mid = (l + r) / 2;
    if (L <= mid) {
        updateAdd(p * 2, l, mid, L, R, k);
    }
    if (R > mid) {
        updateAdd(p * 2 + 1, mid + 1, r, L, R, k);
    }
    pushUp(p);
}
// 区间求和
int query(int p,int l,int r,int L,int R) {
    if (L <= l && r <= R) {
        return tree[p];
    }
    pushDown(p, l, r);
    int mid = (l + r) / 2;
    int ans = 0;
    if (L <= mid) {
        ans += query(p * 2, l, mid, L, R);
        ans %= mod;
    }
    if (R > mid) {
        ans += query(p * 2 + 1, mid + 1, r, L, R);
        ans %= mod;
    }
    return ans;
}
signed main() {
    cin >> n >> m >> mod;
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }
    build(1, 1, n);
    while (m--) {
        int op;
        cin >> op;
        if (op == 1) {
            int x, y, k;
            cin >> x >> y >> k;
            updateMul(1, 1, n, x, y, k % mod);
        } else if (op == 2) {
            int x, y, k;
            cin >> x >> y >> k;
            updateAdd(1, 1, n, x, y, k % mod);
        } else {
            int x, y;
            cin >> x >> y;
            cout << query(1, 1, n, x, y) % mod << '\n';
        }
    }
    return 0;
}
